// FindAll returns all elements of type T in the model.
func FindAll[T Element](model *Model) []T {
    var result []T
    Walk(model, func(elem Element, depth int) bool {
        if t, ok := elem.(T); ok {
            result = append(result, t)
        }
        return true
    })
    return result
}

**visitor.go -- iter.Seq iterators:**

Add iterator-based traversal functions using `iter.Seq` and `iter.Seq2`:

```go
import "iter"

// All returns an iterator over all elements in the model (depth-first).
func All(model *Model) iter.Seq[Element] {
    return func(yield func(Element) bool) {
        for _, elem := range model.Elements {
            if !yieldAll(elem, yield) {
                return
            }
        }
    }
}

// yieldAll yields elem and all its descendants.
func yieldAll(elem Element, yield func(Element) bool) bool {
    if !yield(elem) {
        return false
    }
    for _, child := range elem.Children() {
        if !yieldAll(child, yield) {
            return false
        }
    }
    return true
}

// AllWithDepth returns an iterator with element and depth.
func AllWithDepth(model *Model) iter.Seq2[Element, int] {
    return func(yield func(Element, int) bool) {
        for _, elem := range model.Elements {
            if !yieldAllDepth(elem, 0, yield) {
                return
            }
        }
    }
}

// OfKind returns an iterator over elements of a specific kind.
func OfKind(model *Model, kind ElementKind) iter.Seq[Element] {
    return func(yield func(Element) bool) {
        for elem := range All(model) {
            if elem.Kind() == kind {
                if !yield(elem) {
                    return
                }
            }
        }
    }
}

// OfType returns an iterator over elements of a specific Go type.
func OfType[T Element](model *Model) iter.Seq[T] {
    return func(yield func(T) bool) {
        for elem := range All(model) {
            if t, ok := elem.(T); ok {
                if !yield(t) {
                    return
                }
            }
        }
    }
}
```

This enables composable patterns:
```go
// Collect all parts
parts := slices.Collect(OfType[*Part](model))

// Range directly
for part := range OfType[*Part](model) {
    fmt.Println(part.Name())
}

// Count
count := 0
for _ = range OfKind(model, KindRequirement) {
    count++
}
```

**AddChild pattern -- add generic helper:**

Create a private generic helper to reduce the repetitive AddChild switch patterns:

```go
// addTypedChild is a helper that appends to a typed slice if the child matches type T.
func addTypedChild[T Element](slice *[]T, child Element) {
    if t, ok := child.(T); ok {
        *slice = append(*slice, t)
    }
}
```

Then simplify AddChild methods:
```go
func (p *Part) AddChild(child Element) {
    p.baseElement.addChild(child)
    addTypedChild(&p.attributes, child)
    addTypedChild(&p.parts, child)
    addTypedChild(&p.ports, child)
}
```

Apply this to: Part, Port, Package, Item, State, Constraint, Enumeration, Interface, Action, Verification, Requirement.

**Resolve pattern cleanup:**

The 18+ resolve* methods on Model follow an identical pattern. Create a generic helper:

```go
// resolveRef resolves a typed reference by name.
func resolveRef[T Element](m *Model, ref *Ref[T], context Element) {
    if ref.name == "" || ref.IsResolved() {
        return
    }
    if elem := m.findElement(ref.name, context); elem != nil {
        if t, ok := elem.(T); ok {
            ref.Resolve(t)
        }
    }
}
```

Then simplify each resolve function, e.g.:
```go
func (m *Model) resolvePartRefs(p *Part) {
    if !p.IsDefinition {
        resolveRef(m, &p.TypeRef, p)
    }
}
```

This dramatically reduces the resolve methods. Apply to all 18+ resolve functions.
